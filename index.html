SQL untuk Buat Table

    

      
Jalankan SQL ini dalam Supabase SQL Editor jika belum buat table:

      
CREATE TABLE cs_daily_reports (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  report_date date NOT NULL,
  staff_name text NOT NULL,
  shift text,
  wa1_masuk int DEFAULT 0, wa1_engaged int DEFAULT 0,
  wa2_masuk int DEFAULT 0, wa2_engaged int DEFAULT 0,
  wa3_masuk int DEFAULT 0, wa3_engaged int DEFAULT 0,
  em1_masuk int DEFAULT 0, em1_engaged int DEFAULT 0,
  em2_masuk int DEFAULT 0, em2_engaged int DEFAULT 0,
  notes text,
  created_at timestamptz DEFAULT now()
);
ALTER TABLE cs_daily_reports ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all" ON cs_daily_reports
  FOR ALL USING (true) WITH CHECK (true);
